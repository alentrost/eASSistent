<script type="text/javascript">
(function init_dostopnost() {

    "use strict";

    const $body = $(document.body);
    const $root = $body.find("#dostopnost-widget");
    const $header = $root.find(".dostopnost-header");

    const $temni_kontrast_style  = $("<div>").html('<link rel="stylesheet" type="text/css" href="/css/dostopnost/sogo_contrasts.css">');
    const $svetli_kontrast_style = $("<div>").html('<link rel="stylesheet" type="text/css" href="/css/dostopnost/sogo_contrasts_white.css">');

    let settings = {
        crno_belo:         false,
        temni_kontrast:    false,
        svetli_kontrast:   false,
        velika_pisava:     false,
        podcrtaj_povezave: false,
    };

    const localStorage = window.localStorage || {
        setItem() {},
        getItem() {
            return null;
        },
    };

    function is_open() {
        return !$root.hasClass("closed");
    }

    function set_open(open) {
        $root.toggleClass("closed", !open);
        $header.attr("aria-expanded", open ? "true" : "false");
        $header.attr("aria-label", open ? "Zapri nastavitve dostopnosti" : "Odpri nastavitve dostopnosti");
    }

    function handle_activation(event) {
        if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            $(event.currentTarget).trigger("click");
        }
    }

    $header.on("click", function () {
        set_open(!is_open());
    });

    $header.on("keydown", function (event) {
        if (event.key === "Enter" || event.key === " ") {
            event.preventDefault();
            set_open(!is_open());
        }
        if (event.key === "Escape" && is_open()) {
            event.preventDefault();
            set_open(false);
        }
    });

    $root.on("keydown", "label[data-toggle-dostopnost]", handle_activation);

    $root.on("click", "label[data-toggle-dostopnost]", function () {
        const $label = $(this);
        const key = $label.data("toggle-dostopnost");

        if (key === "clear") {
            for (const prop in settings) {
                settings[prop] = false;
            }
            apply_settings();
            return;
        }

        settings[key] = !settings[key];

        if (key === "temni_kontrast") {
            settings.svetli_kontrast = false;
        }
        if (key === "svetli_kontrast") {
            settings.temni_kontrast = false;
        }

        apply_settings();

        if (window.ga instanceof Function) {
            window.ga("send", {
                hitType: "event",
                eventCategory: "javni urnik",
                eventAction: [$label.text().trim(), settings[key] ? "vklop" : "izklop"].join(" "),
            });
        }
    });

    function apply_settings() {
        $body.toggleClass("sogo_blank_white", settings.crno_belo);
        $body.toggleClass("dostopnost_velika_pisava", settings.velika_pisava);
        $body.toggleClass("dostopnost_podcrtaj_povezave", settings.podcrtaj_povezave);

        if (settings.temni_kontrast) {
            $temni_kontrast_style.appendTo($body);
        } else {
            $temni_kontrast_style.detach();
        }

        if (settings.svetli_kontrast) {
            $svetli_kontrast_style.appendTo($body);
        } else {
            $svetli_kontrast_style.detach();
        }

        for (const setting_key in settings) {
            const $option = $root.find('label[data-toggle-dostopnost="' + setting_key + '"]');
            $option.toggleClass("dostopnost-enabled", settings[setting_key]);
            $option.attr("aria-checked", settings[setting_key] ? "true" : "false");
        }

        localStorage.setItem("dostopnost-settings", JSON.stringify(settings));
    }

    try {
        const stored_settings_json = localStorage.getItem("dostopnost-settings");
        if (stored_settings_json) {
            settings = JSON.parse(stored_settings_json);
            apply_settings();
        }
    } catch {}

    const header_focus_style_id = "public-urnik-focus-styles";
    const header_focus_style_css = "a:focus-visible, button:focus-visible { outline: var(--border-border-size-border-2) solid var(--color-border-color-border-accent) !important; outline-offset: 2px; }";

    function inject_focus_style_into_shadow(shadow_root) {
        if (!shadow_root || shadow_root.getElementById(header_focus_style_id)) {
            return;
        }
        const style = document.createElement("style");
        style.id = header_focus_style_id;
        style.textContent = header_focus_style_css;
        shadow_root.appendChild(style);
    }

    function enhance_header_logo_accessibility() {
        const header = document.querySelector("tim-ui-layout-header");
        if (!header || !header.shadowRoot) {
            return false;
        }

        const logo = header.shadowRoot.querySelector("tim-ui-layout-header-item-logo");
        if (!logo || !logo.shadowRoot) {
            return false;
        }

        const logo_button = logo.shadowRoot.querySelector('button[data-testid="header-item-logo"]');
        const logo_image = logo.shadowRoot.querySelector("img");

        if (logo_button) {
            logo_button.setAttribute("aria-label", "Povezava do prve strani");
        }
        if (logo_image) {
            logo_image.setAttribute("alt", "");
        }

        return Boolean(logo_button);
    }

    function inject_header_focus_styles() {
        const header = document.querySelector("tim-ui-layout-header");
        if (!header || !header.shadowRoot) {
            return false;
        }

        inject_focus_style_into_shadow(header.shadowRoot);

        header.shadowRoot.querySelectorAll("*").forEach(function (element) {
            if (element.shadowRoot) {
                inject_focus_style_into_shadow(element.shadowRoot);
            }
        });

        enhance_header_logo_accessibility();

        const logo = header.shadowRoot.querySelector("tim-ui-layout-header-item-logo");
        return Boolean(logo && logo.shadowRoot && logo.shadowRoot.getElementById(header_focus_style_id));
    }

    function wait_for_header_focus_styles() {
        if (inject_header_focus_styles()) {
            return;
        }
        let attempts = 0;
        const interval = window.setInterval(function () {
            attempts++;
            if (inject_header_focus_styles() || attempts >= 50) {
                window.clearInterval(interval);
            }
        }, 100);
    }

    Promise.all([
        customElements.whenDefined("tim-ui-layout-header"),
        customElements.whenDefined("tim-ui-layout-header-item-logo"),
    ]).then(wait_for_header_focus_styles);
    document.addEventListener("DOMContentLoaded", wait_for_header_focus_styles);
    wait_for_header_focus_styles();
})();
